cmake_minimum_required(VERSION 3.16)
project(lane_shooter C)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

include_directories(${CMAKE_SOURCE_DIR}/include)

file(GLOB_RECURSE SOURCES src/*.c)

add_executable(lane_shooter ${SOURCES})

include(${CMAKE_SOURCE_DIR}/cmake/CPM.cmake)

CPMAddPackage(
  NAME raylib
  GITHUB_REPOSITORY raysan5/raylib
  GIT_TAG 5.5
)

if(raylib_ADDED)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
endif()

target_link_libraries(lane_shooter PRIVATE raylib)

if (WIN32)
    target_link_libraries(lane_shooter PRIVATE winmm)
elseif (UNIX AND NOT APPLE)
    target_link_libraries(lane_shooter PRIVATE m dl pthread GL rt X11)
endif()

set_target_properties(lane_shooter PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}
    RUNTIME_OUTPUT_DIRECTORY_DEBUG ${CMAKE_BINARY_DIR}
    RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}
    RUNTIME_OUTPUT_DIRECTORY_RELWITHDEBINFO ${CMAKE_BINARY_DIR}
    RUNTIME_OUTPUT_DIRECTORY_MINSIZEREL ${CMAKE_BINARY_DIR}
)
